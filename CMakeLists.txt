cmake_minimum_required(VERSION 3.24)
if(CMAKE_HOST_APPLE AND NOT DEFINED CMAKE_OSX_DEPLOYMENT_TARGET)
    set(CMAKE_OSX_DEPLOYMENT_TARGET "13.0" CACHE STRING "Minimum supported macOS")
endif()
project(MafiaPS2Recomp VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
include(CTest)

option(MAFIA_BUILD_RUNTIME "Build PS2Recomp runtime integration" OFF)
option(MAFIA_ENABLE_FFMPEG "Enable original MPEG video decoding" ON)
option(MAFIA_GENERATED_UNITY_BUILD "Combine generated units to reduce header parsing" OFF)
set(MAFIA_GENERATED_UNITY_BATCH_SIZE "32" CACHE STRING "Generated files per unity translation unit")
if(NOT MAFIA_GENERATED_UNITY_BATCH_SIZE MATCHES "^[1-9][0-9]*$")
    message(FATAL_ERROR "MAFIA_GENERATED_UNITY_BATCH_SIZE must be a positive integer")
endif()
set(MAFIA_PS2RECOMP_SOURCE_DIR "" CACHE PATH "Pinned PS2Recomp checkout")
set(MAFIA_GENERATED_DIR "" CACHE PATH "Local upstream generated C++ and generation.json")
set(MAFIA_EXPECTED_UPSTREAM "e5636981a1c9de368f6b71e819c3dc188dfedbf4")

add_library(mafia_host_core STATIC src/verified_elf.cpp src/frame_capture.cpp src/controller_input.cpp)
target_include_directories(mafia_host_core PUBLIC include)
target_compile_options(mafia_host_core PRIVATE -Wall -Wextra -Wpedantic -Wconversion)
if(BUILD_TESTING)
    add_executable(mafia_host_tests tests/cpp/verified_elf_tests.cpp tests/cpp/frame_capture_tests.cpp
        tests/cpp/controller_input_tests.cpp)
    target_link_libraries(mafia_host_tests PRIVATE mafia_host_core)
    add_test(NAME mafia_host_tests COMMAND mafia_host_tests)
    set_tests_properties(mafia_host_tests PROPERTIES TIMEOUT 20)
    add_test(NAME mafia_generated_integrity
        COMMAND "${CMAKE_COMMAND}" "-DPROJECT_ROOT=${CMAKE_CURRENT_SOURCE_DIR}"
            "-DTEST_DIR=${CMAKE_CURRENT_BINARY_DIR}/synthetic-integrity-test"
            -P "${CMAKE_CURRENT_SOURCE_DIR}/tests/cpp/generated_integrity_test.cmake")
    set_tests_properties(mafia_generated_integrity PROPERTIES TIMEOUT 20)
endif()

if(MAFIA_BUILD_RUNTIME)
    if(NOT UNIX)
        message(FATAL_ERROR "The native host currently supports macOS and Linux")
    endif()
    if(APPLE AND NOT CMAKE_OSX_ARCHITECTURES STREQUAL "arm64")
        message(FATAL_ERROR "Build MafiaNative for Apple Silicon with -DCMAKE_OSX_ARCHITECTURES=arm64")
    endif()
    include(cmake/PS2RecompRuntime.cmake)
    add_library(mafia_gs_capture STATIC src/gs_capture.cpp)
    target_link_libraries(mafia_gs_capture PUBLIC mafia_host_core ps2_runtime)
    if(BUILD_TESTING)
        add_executable(mafia_runtime_contract_tests tests/cpp/runtime_contract_tests.cpp)
        target_link_libraries(mafia_runtime_contract_tests PRIVATE ps2_runtime mafia_host_core mafia_gs_capture)
        add_test(NAME mafia_runtime_contract_tests COMMAND mafia_runtime_contract_tests)
        set_tests_properties(mafia_runtime_contract_tests PROPERTIES TIMEOUT 20)
        add_test(NAME mafia_exception_shutdown_timeout
            COMMAND "${CMAKE_COMMAND}" "-DEXECUTABLE=$<TARGET_FILE:mafia_runtime_contract_tests>"
                -P "${CMAKE_CURRENT_SOURCE_DIR}/tests/cpp/unwind_timeout_test.cmake")
        set_tests_properties(mafia_exception_shutdown_timeout PROPERTIES TIMEOUT 12)
        # Exercise the actual backend with isolated authored Raylib input
        # functions. This target opens no window and cannot sample real devices.
        add_executable(mafia_pad_backend_tests tests/cpp/pad_backend_tests.cpp
            "${MAFIA_PS2RECOMP_SOURCE_DIR}/ps2xRuntime/src/lib/ps2_pad.cpp")
        target_include_directories(mafia_pad_backend_tests PRIVATE
            "${MAFIA_PS2RECOMP_SOURCE_DIR}/ps2xRuntime/include")
        target_link_libraries(mafia_pad_backend_tests PRIVATE raylib)
        add_dependencies(mafia_pad_backend_tests mafia_verify_runtime_source)
        add_test(NAME mafia_pad_backend_tests COMMAND mafia_pad_backend_tests)
        set_tests_properties(mafia_pad_backend_tests PROPERTIES TIMEOUT 10)
        include(cmake/NativeHostContract.cmake)
        mafia_add_native_host_contract()
    endif()
    if(MAFIA_GENERATED_DIR)
        find_package(Python3 3.11 REQUIRED COMPONENTS Interpreter)
        get_filename_component(MAFIA_GENERATED_DIR "${MAFIA_GENERATED_DIR}" ABSOLUTE)
        foreach(required generation.json sources.sha256 register_functions.cpp ps2_recompiled_functions.h ps2_recompiled_stubs.h)
            if(NOT EXISTS "${MAFIA_GENERATED_DIR}/${required}")
                message(FATAL_ERROR "Missing generated input: ${MAFIA_GENERATED_DIR}/${required}")
            endif()
        endforeach()
        file(READ "${MAFIA_GENERATED_DIR}/generation.json" generation)
        set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${MAFIA_GENERATED_DIR}/generation.json")
        string(JSON MAFIA_ELF_SHA256 GET "${generation}" elf_sha256)
        string(JSON MAFIA_ELF_ENTRY GET "${generation}" entry_point)
        string(JSON MAFIA_PROFILE_ID GET "${generation}" profile_id)
        string(JSON generated_upstream GET "${generation}" ps2recomp_commit)
        string(JSON sources_manifest_sha256 GET "${generation}" sources_manifest_sha256)
        string(LENGTH "${MAFIA_ELF_SHA256}" digest_length)
        if(NOT digest_length EQUAL 64 OR NOT MAFIA_ELF_SHA256 MATCHES "^[0-9a-f]+$")
            message(FATAL_ERROR "generation.json has an invalid ELF SHA-256")
        endif()
        if(NOT MAFIA_ELF_ENTRY MATCHES "^[0-9]+$" OR MAFIA_ELF_ENTRY GREATER 4294967295)
            message(FATAL_ERROR "generation.json entry_point must be an unsigned 32-bit integer")
        endif()
        if(NOT MAFIA_PROFILE_ID MATCHES "^[a-zA-Z0-9_.-]+$")
            message(FATAL_ERROR "generation.json profile_id must be a simple identifier")
        endif()
        if(NOT generated_upstream STREQUAL MAFIA_EXPECTED_UPSTREAM)
            message(FATAL_ERROR "Generated code uses a different PS2Recomp revision")
        endif()
        string(LENGTH "${sources_manifest_sha256}" manifest_digest_length)
        if(NOT manifest_digest_length EQUAL 64 OR NOT sources_manifest_sha256 MATCHES "^[0-9a-f]+$")
            message(FATAL_ERROR "generation.json has an invalid source manifest SHA-256")
        endif()
        if(MAFIA_ENABLE_FFMPEG)
            set(MAFIA_FFMPEG_BOOL true)
        else()
            set(MAFIA_FFMPEG_BOOL false)
        endif()
        configure_file(cmake/build_profile.hpp.in generated/mafia_build_profile.hpp @ONLY)
        file(GLOB generated_sources CONFIGURE_DEPENDS "${MAFIA_GENERATED_DIR}/*.cpp")
        add_executable(MafiaNative src/native_main.cpp ${generated_sources})
        add_custom_target(mafia_verify_generated
            COMMAND "${CMAKE_COMMAND}" "-DGENERATED_DIR=${MAFIA_GENERATED_DIR}"
                "-DEXPECTED_MANIFEST_SHA256=${sources_manifest_sha256}"
                -P "${CMAKE_CURRENT_SOURCE_DIR}/cmake/VerifyGenerated.cmake"
            VERBATIM)
        add_custom_target(mafia_verify_upstream
            COMMAND "${CMAKE_COMMAND}" "-DUPSTREAM_DIR=${MAFIA_PS2RECOMP_SOURCE_DIR}"
                "-DPATCH_DIR=${CMAKE_CURRENT_SOURCE_DIR}/patches"
                "-DGENERATION_FILE=${MAFIA_GENERATED_DIR}/generation.json"
                "-DEXPECTED_REVISION=${MAFIA_EXPECTED_UPSTREAM}"
                -P "${CMAKE_CURRENT_SOURCE_DIR}/cmake/VerifyUpstream.cmake"
            COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tools/verify_upstream.py"
                "${MAFIA_PS2RECOMP_SOURCE_DIR}"
            VERBATIM)
        add_dependencies(MafiaNative mafia_verify_generated)
        add_dependencies(MafiaNative mafia_verify_upstream)
        if(MAFIA_GENERATED_UNITY_BUILD)
            set_target_properties(MafiaNative PROPERTIES UNITY_BUILD ON
                UNITY_BUILD_BATCH_SIZE "${MAFIA_GENERATED_UNITY_BATCH_SIZE}")
            set_source_files_properties(src/native_main.cpp "${MAFIA_GENERATED_DIR}/register_functions.cpp"
                PROPERTIES SKIP_UNITY_BUILD_INCLUSION ON)
        endif()
        target_include_directories(MafiaNative PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated" "${MAFIA_GENERATED_DIR}")
        target_link_libraries(MafiaNative PRIVATE mafia_host_core ps2_runtime mafia_gs_capture)
        target_compile_options(MafiaNative PRIVATE -fno-fast-math -ffp-contract=off)
        if(APPLE)
            set_target_properties(MafiaNative PROPERTIES MACOSX_BUNDLE TRUE
                MACOSX_BUNDLE_INFO_PLIST "${CMAKE_CURRENT_SOURCE_DIR}/cmake/Info.plist.in")
            install(TARGETS MafiaNative BUNDLE DESTINATION .)
        else()
            install(TARGETS MafiaNative RUNTIME DESTINATION bin)
        endif()
    else()
        message(STATUS "No game corpus selected: only runtime contract tests are available")
    endif()
endif()
